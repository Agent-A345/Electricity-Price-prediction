@st.cache_data
def create_dataset():

    np.random.seed(42)

    dates = pd.date_range(
        "2023-01-01",
        "2025-12-31",
        freq="D"
    )

    day = np.arange(len(dates))

    weekly = 15 * np.sin(
        2 * np.pi * day / 7
    )

    yearly = 12 * np.sin(
        2 * np.pi * day / 365
    )

    noise = np.random.normal(
        0,
        2,
        len(dates)
    )

    consumption = (
        120
        + weekly
        + yearly
        + noise
    )

    price = (
        8
        + 0.03 * (consumption - 120)
        + np.random.normal(
            0,
            0.15,
            len(dates)
        )
    )

    price = np.maximum(
        price,
        5
    )

    df = pd.DataFrame({
        "Date": dates,
        "Consumption": consumption,
        "Price_per_kWh": price
    })

    return df


@st.cache_resource(show_spinner=False)
def train_model():

    df = create_dataset()

    scaler = MinMaxScalerr()

    scaled_consumption = scaler.fit_transform(
        df[["Consumption"]]
    ).flatten()

    LOOKBACK = 30
    FORECAST_DAYS = 30

    X = []
    y = []

    for i in range(
        len(scaled_consumption)
        - LOOKBACK
        - FORECAST_DAYS
        + 1
    ):

        X.append(
            scaled_consumption[
                i:i + LOOKBACK
            ]
        )

        y.append(
            scaled_consumption[
                i + LOOKBACK:
                i + LOOKBACK + FORECAST_DAYS
            ]
        )

    X = np.array(X)
    y = np.array(y)

    X = X.reshape(
        X.shape[0],
        LOOKBACK,
        1
    )

    split = int(
        len(X) * 0.80
    )

    X_train = X[:split]
    y_train = y[:split]

    model = Sequential([
        Input(shape=(30, 1)),

        LSTM(
            64,
            return_sequences=True
        ),

        Dropout(0.10),

        LSTM(32),

        Dropout(0.10),

        Dense(
            32,
            activation="relu"
        ),

        Dense(30)
    ])

    model.compile(
        optimizer="adam",
        loss="mse"
    )

    early_stop = EarlyStopping(
        monitor="val_loss",
        patience=5,
        restore_best_weights=True
    )

    model.fit(
        X_train,
        y_train,
        epochs=40,
        batch_size=32,
        validation_split=0.10,
        callbacks=[early_stop],
        verbose=0
    )

    return (
        model,
        scaler,
        scaled_consumption
    )
